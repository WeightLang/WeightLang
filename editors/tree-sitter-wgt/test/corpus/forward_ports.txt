wgt 0.5
model ExplicitHidden
# Two features and two hidden channels use independent host-provided vectors.
# The initialized gain is shared by the prediction and next-hidden equations;
# initialization is explicit instance creation, not compilation or training.
parameter gain: f32[2] = full(0.5)

# Both outputs retain their logical vector shape. Hidden state belongs to the
# host: pass next_hidden into a later call, or detach it at a training boundary.
# No fn loss is declared: a host loss/output seed chooses which outputs matter.
fn forward(x: f32[2], hidden: f32[2]) -> (f32[2], f32[2]) {
    let prediction = x * gain + hidden;
    let next_hidden = hidden * gain + x;
    return (prediction, next_hidden);
}

examples/native_training/dynamic_hidden/model.wgt
================================================================================

wgt 0.5
dim B = 2
model DynamicHidden
# B is an activation-only row axis. This scalar gate is shared by every row and
# every call; changing B must preserve its value, identity and optimizer history.
parameter gain: f32[1] = [0.5]

# x and hidden have matching logical rows. The host owns hidden and explicitly
# chooses which output contributes loss; neither output is persistent state.
# Repeated calls can have different B, while each return retains its own shape.
fn forward(x: f32[B,1], hidden: f32[B,1]) -> (f32[B,1], f32[B,1]) {
    let prediction = scale(x, gain) + hidden;
    let next_hidden = scale(hidden, gain) + x;
    return (prediction, next_hidden);
}

}
